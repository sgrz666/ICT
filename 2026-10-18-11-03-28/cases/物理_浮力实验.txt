<h3>原理</h3>
<ul>
  <li>浸在液体中的物体受到竖直向上的<b>浮力</b>，大小等于它<b>排开</b>的液体所受的重力，即 F浮 = ρ液 g V排——这就是阿基米德原理。</li>
  <li>用弹簧测力计悬挂物块：空气中示数等于重力 G，浸入后示数变小，减少的部分就是浮力，所以 F浮 = G - F示，这叫称重法。</li>
  <li>物块浸入后要占据液体原本的空间，量筒内液面会上升；上升的体积正好等于排开液体的体积 V排。</li>
  <li>浮沉由 ρ物 与 ρ液 的关系决定：ρ物 &lt; ρ液 上浮至漂浮（此时 F浮 = G）；ρ物 = ρ液 悬浮；ρ物 &gt; ρ液 下沉沉底。</li>
</ul>
<h3>易错点</h3>
<ul>
  <li>以为"浸得越深浮力越大"：浸没以后 V排 不再增加，浮力与深度无关。</li>
  <li>把物块的体积当成 V排：没有浸没时 V排 只是物块浸入那部分的体积。</li>
  <li>忘记单位换算：1 cm³ = 1×10⁻⁶ m³，1 g/cm³ = 1000 kg/m³，代入式里必须统一成国际单位。</li>
  <li>误以为漂浮时浮力会"不够用"：漂浮时二力平衡，F浮 恰好等于 G，一点也不会少。</li>
</ul>
<h3>探究问题</h3>
<ol>
  <li>把物块完全浸没后继续加深，测力计示数还变吗？这说明浮力和深度有关吗？</li>
  <li>保持物块不变，把液体从水换成浓盐水，F浮 怎么变？此时物块是上浮还是下沉？</li>
  <li>把 ρ物 调到 0.6 g/cm³ 再切到"自由浮沉"，物块最终露出液面的体积占总体积的几分之几？</li>
</ol>
<h3>结论</h3>
<p>浮力大小只由液体密度和排开液体的体积决定，与物体浸没的深度无关；称重法把浮力变成两次读数之差，便于测量；物体上浮、悬浮还是下沉，取决于物体密度与液体密度的大小关系，漂浮时 F浮 = G。</p>
<h3>注意事项</h3>
<ul>
  <li>本页为虚拟实验，读数由 F浮 = ρ液 g V排 实时计算（g 取 10 N/kg），不是模型估算。</li>
  <li>弹簧测力计量程 0~5 N：物块密度与体积调到很大时示数可能超出量程，实际实验应换用量程更大的测力计。</li>
  <li>真实实验中物块要完全浸没且不接触容器底与侧壁，否则示数会偏小。</li>
  <li>虚拟演示不能替代真实操作，实验技能与误差分析仍需在实验室完成。</li>
</ul>

(function () {
  'use strict';

  var cv = document.getElementById('cv');
  if (!cv) return;
  var ctx = cv.getContext('2d');

  /* ---------------- 画布与场景尺寸（设计坐标 780×600） ---------------- */
  var DW = 780, DH = 600;
  var PX = 14;                 // 1 cm = 14 px
  var G_ACC = 10;              // N/kg，教材取 10
  var CX = 300;                // 量筒中心线
  var Y_BOT = 512;             // 量筒底（台面）
  var BEAKER_H = 16;           // 量筒高度 cm
  var Y_TOP = Y_BOT - BEAKER_H * PX;
  var R_IN = 62, R_OUT = 69;
  var AB = 62.5;               // 量筒截面积 cm²（16 cm ≈ 1000 mL）
  var H0 = 9.6;                // 初始液面高度 cm（600 mL）
  var TABLE_Y = 512;
  var STAND_X = 712;
  var STRING = 80;             // 细线长度 px
  var BODY_H = 150;            // 测力计刻度区高度 px

  var view = { yaw: 0.35, k: 0.26, s: 1, ox: 0, oy: 0, w: 0, h: 0, dpr: 1 };

  var DEF = { rhoL: 1.0, rho: 2.7, vol: 100, sub: 0.6 };
  var S = {
    rhoL: DEF.rhoL, rho: DEF.rho, vol: DEF.vol, sub: DEF.sub,
    mode: 'hang', yb: 0, v: 0, playing: true, soak: false, t: 0, settle: 0, stepDir: 1
  };

  /* ---------------- 材质 / 液体外观 ---------------- */
  function matOf(r) {
    var list = [
      [0.35, '软木', [216, 189, 143]],
      [0.65, '木块', [198, 154, 99]],
      [0.95, '冰块', [205, 233, 247]],
      [1.15, '塑料', [127, 192, 163]],
      [1.8, '蜡块', [236, 217, 168]],
      [2.85, '铝块', [183, 193, 203]],
      [9.99, '铁块', [152, 162, 174]]
    ];
    for (var i = 0; i < list.length; i++) {
      if (r <= list[i][0] + 1e-9) return { name: list[i][1], rgb: list[i][2] };
    }
    return { name: '铁块', rgb: [152, 162, 174] };
  }

  function liqOf(r) {
    if (r < 0.9) return { name: '酒精', top: [230, 243, 252], bot: [170, 212, 235] };
    if (r < 1.1) return { name: '水', top: [201, 232, 250], bot: [140, 199, 232] };
    if (r < 1.25) return { name: '盐水', top: [198, 232, 245], bot: [126, 191, 209] };
    return { name: '浓盐水', top: [190, 228, 242], bot: [110, 179, 199] };
  }

  /* ---------------- 颜色工具 ---------------- */
  function css(c, a) {
    if (a === undefined) return 'rgb(' + (c[0] | 0) + ',' + (c[1] | 0) + ',' + (c[2] | 0) + ')';
    return 'rgba(' + (c[0] | 0) + ',' + (c[1] | 0) + ',' + (c[2] | 0) + ',' + a + ')';
  }
  function shade(c, k) {
    var o = [c[0], c[1], c[2]];
    for (var i = 0; i < 3; i++) {
      if (k >= 0) o[i] = o[i] + (255 - o[i]) * k;
      else o[i] = o[i] * (1 + k);
    }
    return o;
  }
  function mix(c1, c2, t) {
    return [c1[0] + (c2[0] - c1[0]) * t,
            c1[1] + (c2[1] - c1[1]) * t,
            c1[2] + (c2[2] - c1[2]) * t];
  }
  var WATER_TINT = [108, 168, 204];

  /* ---------------- 物理计算 ---------------- */
  function derived() {
    var a = Math.cbrt(S.vol);              // 立方体边长 cm
    var m = S.rho * S.vol / 1000;          // kg
    var G = m * G_ACC;                     // N
    var A = a * a;                         // 横截面积 cm²
    var s, Vsub, h, Fb, T, yb;

    if (S.mode === 'hang') {
      s = S.sub * a;
      Vsub = A * s;
      h = H0 + Vsub / AB;
      Fb = S.rhoL * 1000 * G_ACC * (Vsub / 1e6);
      T = Math.max(0, G - Fb);
      yb = h - s;
    } else {
      s = Math.min(a, Math.max(0, H0 - S.yb + (A * a) / AB * 0.5));
      for (var i = 0; i < 5; i++) {
        h = H0 + A * s / AB;
        s = Math.min(a, Math.max(0, h - S.yb));
      }
      Vsub = A * s;
      Fb = S.rhoL * 1000 * G_ACC * (Vsub / 1e6);
      T = 0;
      yb = S.yb;
    }
    return { a: a, m: m, G: G, A: A, s: s, Vsub: Vsub, h: h, Fb: Fb, T: T, yb: yb };
  }

  function stepPhysics(dt) {
    var n = 8, hh = dt / n;
    for (var i = 0; i < n; i++) {
      var d = derived();
      var c = 8 * d.m + 0.6;                       // 阻尼 N/(m/s)
      var F = d.Fb - d.G - c * S.v;
      var acc = F / Math.max(d.m, 0.01);
      S.v += acc * hh;
      S.yb += S.v * 100 * hh;                      // m/s → cm/s
      if (S.yb < 0) { S.yb = 0; S.v = 0; }
      if (S.yb > 17) { S.yb = 17; S.v = 0; }
    }
  }

  function stateOf(d) {
    var eps = 0.02;
    if (S.mode === 'hang') {
      if (S.sub <= 0.001) return { txt: '未入水：示数等于物块重力', cls: '' };
      if (d.s >= d.a - 1e-6) {
        if (d.Fb > d.G + 1e-6) return { txt: '浸没但仍上浮：细线松弛，示数为 0', cls: 'is-warn' };
        return { txt: '浸没：V排 = V物，浮力达到最大', cls: '' };
      }
      return { txt: '部分浸入：V排 < V物，示数随浸入减小', cls: '' };
    }
    if (S.rho > S.rhoL + eps) {
      return Math.abs(S.v) > 0.03 ? { txt: '下沉中：F浮 < G', cls: 'is-sink' } : { txt: '沉底：ρ物 > ρ液，F浮 < G', cls: 'is-sink' };
    }
    if (S.rho < S.rhoL - eps) {
      return Math.abs(S.v) > 0.03 ? { txt: '上浮中：F浮 > G', cls: '' } : { txt: '漂浮：ρ物 < ρ液，F浮 = G', cls: '' };
    }
    return { txt: '悬浮：ρ物 ≈ ρ液，可停在任意深度', cls: '' };
  }

  /* ---------------- 画布适配 ---------------- */
  function fit() {
    var box = cv.parentNode;
    var w = box.clientWidth || 700;
    var h = box.clientHeight || 460;
    view.dpr = window.devicePixelRatio || 1;
    view.w = w; view.h = h;
    cv.width = Math.max(1, Math.round(w * view.dpr));
    cv.height = Math.max(1, Math.round(h * view.dpr));
    cv.style.width = w + 'px';
    cv.style.height = h + 'px';
    view.s = Math.min(w / DW, h / DH);
    view.ox = (w - DW * view.s) / 2;
    view.oy = (h - DH * view.s) / 2;
  }

  function begin() {
    ctx.setTransform(view.dpr, 0, 0, view.dpr, 0, 0);
    ctx.clearRect(0, 0, view.w, view.h);
    ctx.save();
    ctx.translate(view.ox, view.oy);
    ctx.scale(view.s, view.s);
    ctx.lineJoin = 'round';
    ctx.lineCap = 'round';
  }

  function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.arcTo(x + w, y, x + w, y + h, r);
    ctx.arcTo(x + w, y + h, x, y + h, r);
    ctx.arcTo(x, y + h, x, y, r);
    ctx.arcTo(x, y, x + w, y, r);
    ctx.closePath();
  }

  function bodyPath(r, ry) {
    ctx.beginPath();
    ctx.moveTo(CX - r, Y_TOP);
    ctx.lineTo(CX - r, Y_BOT - 12);
    ctx.quadraticCurveTo(CX - r, Y_BOT, CX - r + 14, Y_BOT);
    ctx.lineTo(CX + r - 14, Y_BOT);
    ctx.quadraticCurveTo(CX + r, Y_BOT, CX + r, Y_BOT - 12);
    ctx.lineTo(CX + r, Y_TOP);
    ctx.closePath();
  }

  /* ---------------- 背景实验台 ---------------- */
  function drawBG() {
    var g = ctx.createLinearGradient(0, 0, 0, DH);
    g.addColorStop(0, '#fcfdff');
    g.addColorStop(0.55, '#f2f8fd');
    g.addColorStop(1, '#e7f0f8');
    ctx.fillStyle = g;
    ctx.fillRect(0, 0, DW, DH);

    var rg = ctx.createRadialGradient(200, 70, 30, 200, 70, 560);
    rg.addColorStop(0, 'rgba(255,255,255,0.95)');
    rg.addColorStop(1, 'rgba(255,255,255,0)');

    ctx.fillRect(0, 0, DW, DH);

    tg.addColorStop(0.25, '#e2d9ca');
    tg.addColorStop(1, '#cfc4b3');

    sh.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = sh;
    ctx.fillRect(0, TABLE_Y, DW, 26);
  }

  function drawStand() {
    // 底座
    var bg = ctx.createLinearGradient(0, 494, 0, 514);
    bg.addColorStop(0, '#a8b3be');
    bg.addColorStop(0.45, '#7f8b98');
    bg.addColorStop(1, '#5f6a76');

    ctx.stroke();
    ctx.fillStyle = 'rgba(255,255,255,0.35)';

    rg.addColorStop(0, '#9aa6b2');
    rg.addColorStop(0.35, '#e3e9ef');
    rg.addColorStop(0.7, '#b3bec9');
    rg.addColorStop(1, '#78848f');
    ctx.fillStyle = rg;
    roundRect(STAND_X - 5, 84, 10, 414, 4);
    ctx.fill();
    ctx.strokeStyle = 'rgba(80,95,110,0.35)';
    ctx.stroke();
  }

  function drawBeakerBack(ry) {
    // 台下阴影
    ctx.fillStyle = 'rgba(52,72,92,0.10)';
    ctx.beginPath();
    ctx.ellipse(CX + 8, TABLE_Y + 5, R_OUT + 34, 11, 0, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = 'rgba(52,72,92,0.10)';
    ctx.beginPath();
    ctx.ellipse(CX + 3, TABLE_Y + 2, R_OUT + 12, 7, 0, 0, Math.PI * 2);
    ctx.fill();

    // 玻璃筒身（背面）
    bodyPath(R_IN, ry);
    var g = ctx.createLinearGradient(CX - R_IN, 0, CX + R_IN, 0);
    g.addColorStop(0, 'rgba(224,241,252,0.85)');
    g.addColorStop(0.28, 'rgba(246,252,255,0.45)');
    g.addColorStop(0.62, 'rgba(238,249,255,0.30)');
    g.addColorStop(1, 'rgba(210,234,249,0.75)');
    ctx.fillStyle = g;
    ctx.fill();

    // 筒口后半圈（在物块之后会被前半圈盖住）
    ctx.beginPath();
    ctx.ellipse(CX, Y_TOP, R_OUT, ry * R_OUT / R_IN, 0, Math.PI, Math.PI * 2);
    ctx.strokeStyle = 'rgba(146,182,210,0.95)';
    ctx.lineWidth = 2.6;
    ctx.stroke();
  }

  function drawLiquid(d, ry) {
    var yW = Y_BOT - d.h * PX;
    var lc = liqOf(S.rhoL);

    ctx.save();
    bodyPath(R_IN, ry);
    ctx.clip();
    var g = ctx.createLinearGradient(0, yW, 0, Y_BOT);
    g.addColorStop(0, css(lc.top));
    g.addColorStop(0.55, css(mix(lc.top, lc.bot, 0.55)));
    g.addColorStop(1, css(lc.bot));
    ctx.fillStyle = g;
    ctx.fillRect(CX - R_IN, yW, R_IN * 2, Y_BOT - yW);

    // 后壁折射亮带
    ctx.fillStyle = 'rgba(255,255,255,0.18)';
    ctx.fillRect(CX - R_IN + 6, yW, 12, Y_BOT - yW);

    // 气泡（自由浮沉时更明显）
    if (S.mode === 'free') {
      for (var i = 0; i < 6; i++) {
        var ph = (S.t * 0.35 + i * 0.37) % 1;
        var bx = CX - 30 + ((i * 37) % 60);
        var by = Y_BOT - 6 - ph * (Y_BOT - yW - 10);
        var br = 1.2 + (i % 3) * 0.7;
        ctx.beginPath();
        ctx.arc(bx + Math.sin(S.t * 1.5 + i) * 3, by, br, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(255,255,255,' + (0.35 * (1 - ph)) + ')';
        ctx.fill();
      }
    }
    ctx.restore();

    // 液面（上表面椭圆）
    ctx.beginPath();
    ctx.ellipse(CX, yW, R_IN, ry, 0, 0, Math.PI * 2);
    var sg = ctx.createLinearGradient(CX - R_IN, yW - ry, CX + R_IN, yW + ry);
    sg.addColorStop(0, 'rgba(255,255,255,0.98)');
    sg.addColorStop(0.55, css(mix(lc.top, [255, 255, 255], 0.45)));
    sg.addColorStop(1, css(lc.top));
    ctx.fillStyle = sg;
    ctx.fill();
    ctx.strokeStyle = 'rgba(116,170,206,0.9)';
    ctx.lineWidth = 1.3;
    ctx.stroke();

    // 弯月面
    ctx.beginPath();
    ctx.ellipse(CX, yW + 1.4, R_IN - 3.5, Math.max(2, ry * 0.8), 0, 0, Math.PI * 2);
    ctx.strokeStyle = 'rgba(255,255,255,0.9)';
    ctx.lineWidth = 1.6;
    ctx.stroke();

    // 水面波纹
    for (var w = 0; w < 3; w++) {
      var amp = 0.9 + w * 0.5;
      var yy = yW + 3 + w * 3.2;
      ctx.beginPath();
      for (var x = -R_IN + 6; x <= R_IN - 6; x += 4) {
        var py = yy + Math.sin(x * 0.16 + S.t * 1.7 + w) * amp * 0.5;
        if (x === -R_IN + 6) ctx.moveTo(CX + x, py);
        else ctx.lineTo(CX + x, py);
      }
      ctx.strokeStyle = 'rgba(255,255,255,' + (0.34 - w * 0.08) + ')';
      ctx.lineWidth = 1.1;
      ctx.stroke();
    }
  }

  /* ---------------- 物块（斜投影立方体） ---------------- */
  function drawCube(bottomY, ap, ox, oy, col, tint) {
    var x0 = CX - ap / 2, x1 = CX + ap / 2;
    var y0 = bottomY - ap, y1 = bottomY;

    function fill(pathFn, base, light) {
      pathFn();
      var c = light ? shade(base, light) : base;
      if (tint) c = mix(c, WATER_TINT, tint);
      ctx.fillStyle = css(c);
      ctx.fill();
      ctx.strokeStyle = 'rgba(38,58,78,' + (tint ? 0.28 : 0.42) + ')';
      ctx.lineWidth = 1.1;
      ctx.stroke();
    }

    // 右侧面
    fill(function () {
      ctx.beginPath();
      ctx.moveTo(x1, y0);
      ctx.lineTo(x1 + ox, y0 - oy);
      ctx.lineTo(x1 + ox, y1 - oy);
      ctx.lineTo(x1, y1);
      ctx.closePath();
    }, col, -0.26);

    // 顶面
    fill(function () {
      ctx.beginPath();
      ctx.moveTo(x0, y0);
      ctx.lineTo(x1, y0);
      ctx.lineTo(x1 + ox, y0 - oy);
      ctx.lineTo(x0 + ox, y0 - oy);
      ctx.closePath();
    }, col, 0.38);

    // 正面
    var g = ctx.createLinearGradient(x0, y0, x1, y1);
    var c1 = shade(col, 0.14), c2 = shade(col, -0.12);
    if (tint) { c1 = mix(c1, WATER_TINT, tint); c2 = mix(c2, WATER_TINT, tint); }
    g.addColorStop(0, css(c1));
    g.addColorStop(1, css(c2));
    ctx.fillStyle = g;
    ctx.fillRect(x0, y0, ap, ap);
    ctx.strokeStyle = 'rgba(38,58,78,' + (tint ? 0.30 : 0.45) + ')';
    ctx.lineWidth = 1.1;
    ctx.strokeRect(x0, y0, ap, ap);

    // 正面高光
    var hg = ctx.createLinearGradient(x0, y0, x0, y1);
    hg.addColorStop(0, 'rgba(255,255,255,0.28)');
    hg.addColorStop(0.45, 'rgba(255,255,255,0.06)');
    hg.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = hg;
    ctx.fillRect(x0 + 1, y0 + 1, ap - 2, ap * 0.55);
  }

  function drawObject(d, ry) {
    var ap = d.a * PX;
    var yW = Y_BOT - d.h * PX;
    var bottom;
    if (S.mode === 'hang') bottom = yW + d.s * PX;
    else bottom = Y_BOT - d.yb * PX;

    var mat = matOf(S.rho);
    var ox = 0.34 * ap * Math.cos(view.yaw);
    var oy = 0.26 * ap;

    // 水中投影
    if (S.mode === 'free') {
      ctx.beginPath();
      ctx.ellipse(CX + ox * 0.4, Y_BOT - 4, ap * 0.62, ap * 0.14, 0, 0, Math.PI * 2);
      ctx.fillStyle = 'rgba(40,70,95,0.14)';
      ctx.fill();
    }

    // 水线以上部分
    ctx.save();
    ctx.beginPath();
    ctx.rect(0, 0, DW, yW);
    ctx.clip();
    drawCube(bottom, ap, ox, oy, mat.rgb, 0);
    ctx.restore();

    // 水线以下部分（带折射偏移与蓝色调）
    ctx.save();
    ctx.beginPath();
    ctx.rect(0, yW, DW, DH - yW);
    ctx.clip();
    ctx.translate(2.4, 0);
    drawCube(bottom, ap, ox, oy, mat.rgb, 0.30);
    ctx.restore();

    // 水线处的前半圈液面
    ctx.beginPath();
    ctx.ellipse(CX, yW, R_IN, ry, 0, 0, Math.PI);
    ctx.strokeStyle = 'rgba(255,255,255,0.75)';
    ctx.lineWidth = 1.4;
    ctx.stroke();

    return { top: bottom - ap - oy, mid: bottom - ap * 0.6 };
  }

      ctx.beginPath();

      ctx.strokeStyle = 'rgba(255,255,255,' + (0.5 * (1 - ph)) + ')';
      ctx.lineWidth = 1.2;
      ctx.stroke();
    }
  }

  function render() {
    var d = derived();
    var a = d.a;

    var Gs = Math.round(d.G * 100) / 100;
    var Fbs = Math.round(d.Fb * 100) / 100;
    var Ts = Math.max(0, Math.round((Gs - Fbs) * 100) / 100);
    var VsubShow = Math.round(d.Vsub * 10) / 10;

    setText('v-rho', S.rho.toFixed(2));
    setText('v-mat', matOf(S.rho).name);
    setText('v-vol', String(S.vol));
    setText('v-sub', (S.sub * a).toFixed(1));
    setText('v-liq', liqOf(S.rhoL).name + ' ' + S.rhoL.toFixed(2) + ' g/cm³');

    setText('r-fb', Fbs.toFixed(2));
    setText('r-ft', (S.mode === 'hang' ? Ts : 0).toFixed(2));
    setText('r-vp', VsubShow.toFixed(1));
    setText('r-s', d.s.toFixed(1));

    var st = stateOf(d);
    var stateEl = document.getElementById('state');
    stateEl.className = 'exp-state' + (st.cls ? ' ' + st.cls : '');
    setText('state-txt', st.txt);

    // 代入式：所有数字都用「页面上显示的同一套」，保证读者能自己复算
    var lines = [];
    if (d.Vsub > 0.01) {
      var vm3 = Number((d.Vsub / 1e6).toPrecision(4));
      lines.push('F浮 = ρ液 g V排 = ' + (S.rhoL * 1000).toFixed(0) + ' kg/m³ × 10 N/kg × ' + String(vm3) + ' m³ = ' + Fbs.toFixed(2) + ' N');
    } else {
      lines.push('V排 = 0 cm³（物块尚未排开液体），F浮 = 0.00 N');
    }
    lines.push('G = m g = ' + (d.m).toFixed(3) + ' kg × 10 N/kg = ' + Gs.toFixed(2) + ' N');
    if (S.mode === 'hang') {
      if (Fbs < Gs - 0.005) {
        lines.push('F示 = G - F浮 = ' + Gs.toFixed(2) + ' N - ' + Fbs.toFixed(2) + ' N = ' + Ts.toFixed(2) + ' N');
      } else if (Fbs > Gs + 0.005) {
        lines.push('F浮 - G = ' + Fbs.toFixed(2) + ' N - ' + Gs.toFixed(2) + ' N = ' + (Math.round((Fbs - Gs) * 100) / 100).toFixed(2) + ' N（净力向上，细线松弛，F示 = 0 N）');
      }
    } else {
      lines.push('ρ物 = ' + S.rho.toFixed(2) + ' g/cm³，ρ液 = ' + S.rhoL.toFixed(2) + ' g/cm³ → ' + (S.rho > S.rhoL + 0.02 ? '下沉' : (S.rho < S.rhoL - 0.02 ? '上浮' : '悬浮')));
    }
    var fmlEl = document.getElementById('fml');
    var fmlHtml = lines.join('<br>');
    if (fmlEl.getAttribute('data-txt') !== fmlHtml) {
      fmlEl.innerHTML = fmlHtml;
      fmlEl.setAttribute('data-txt', fmlHtml);
    }

    var hint = document.getElementById('hint');
    var alertBox = document.getElementById('alert');
    if (S.mode === 'hang') {
      if (d.s >= a - 1e-6) {
        setText('hint', '物块已完全浸没：再加深深度，V排 不再增加，F浮 保持不变——浮力与深度无关。');
      } else if (d.s <= 1e-6) {
        setText('hint', '物块下表面刚接触液面，V排 = 0，此时测力计示数等于物块重力。');
      } else {
        setText('hint', '物块正在进入液体：浸得越深，排开的液体越多，F浮 越大，测力计示数越小。');
      }
      if (d.Fb >= d.G) {
        alertBox.hidden = false;
        alertBox.textContent = 'F浮 ≥ G：物块本身会浮起来，细线已经松弛，测力计示数为 0。这说明该物块密度小于液体密度。';
      } else {
        alertBox.hidden = true;
      }
    } else {
      setText('hint', '松手后由合力决定运动：F浮 > G 上浮，F浮 < G 下沉，F浮 = G 悬浮。最终露出液面的比例由 ρ物 / ρ液 决定。');
      alertBox.hidden = true;
    }

    var tip = document.getElementById('tip');
    setText('tip', '装置：弹簧测力计（0~5 N）· 量筒（1000 mL）· ' + liqOf(S.rhoL).name + ' · 物块（' + matOf(S.rho).name + '）');

    window.__DATA__ = {
      experiment_id: 'buoyancy_measure',
      rho_liquid: S.rhoL,
      rho_object: S.rho,
      volume: S.vol,
      V_displaced: Number(d.Vsub.toFixed(2)),
      F_buoy: Number(d.Fb.toFixed(3)),
      F_spring: Number((S.mode === 'hang' ? d.T : 0).toFixed(3)),
      G: Number(d.G.toFixed(3)),
      immersion_depth: Number(d.s.toFixed(3)),
      liquid_level: Number(d.h.toFixed(3)),
      state: st.txt,
      mode: S.mode,
      checks: {
        '浮力公式复算': Math.abs(d.Fb - S.rhoL * 1000 * G_ACC * d.Vsub / 1e6) < 1e-6,
        '排开体积不超物体体积': d.Vsub <= S.vol + 1e-6,
        '称重法自洽': S.mode !== 'hang' || Math.abs(Math.max(0, d.G - d.Fb) - d.T) < 1e-6
      }
    };
    window.__CONFIG__ = {
      scene_id: 'buoyancy_measure',
      scene_engine: 'measurement',
      subject: '物理',
      experiment_id: '浮力测量'
    };

    draw(d);
  }

  /* ---------------- 实验记录 ---------------- */
  var rowsEl = document.getElementById('rows');
  function record() {
    var d = derived();
    var Gs = Math.round(d.G * 100) / 100;
    var Fbs = Math.round(d.Fb * 100) / 100;
    var Ts = Math.max(0, Math.round((Gs - Fbs) * 100) / 100);
    var st = stateOf(d);
    var cells = [
      String(rowsEl.children.length + 1),
      liqOf(S.rhoL).name,
      S.rhoL.toFixed(2),
      S.rho.toFixed(2),
      String(S.vol),
      (Math.round(d.Vsub * 10) / 10).toFixed(1),
      Fbs.toFixed(2),
      (S.mode === 'hang' ? Ts : 0).toFixed(2),
      st.txt
    ];
    var tr = document.createElement('tr');
    for (var i = 0; i < cells.length; i++) {
      var td = document.createElement('td');
      if (i === 6 || i === 7) td.className = 'is-strong';
      td.textContent = cells[i];
      tr.appendChild(td);
    }
    rowsEl.appendChild(tr);
  }

  function csv() {
    var lines = ['序号,液体,ρ液(g/cm³),ρ物(g/cm³),V物(cm³),V排(cm³),F浮(N),F示(N),状态'];
    for (var i = 0; i < rowsEl.children.length; i++) {
      var tds = rowsEl.children[i].children, arr = [];
      for (var j = 0; j < tds.length; j++) {
        var v = tds[j].textContent;
        arr.push(/[",\n]/.test(v) ? '"' + v.replace(/"/g, '""') + '"' : v);
      }
      lines.push(arr.join(','));
    }
    var blob = new Blob(['\ufeff' + lines.join('\r\n')], { type: 'text/csv;charset=utf-8' });
    var url = URL.createObjectURL(blob);
    var a = document.createElement('a');
    a.href = url;
    a.download = '浮力实验记录.csv';
    a.click();
    setTimeout(function () { URL.revokeObjectURL(url); }, 2000);
  }

  /* ---------------- 控件 ---------------- */
  function syncControls() {
    document.getElementById('rg-rho').value = String(S.rho);
    document.getElementById('rg-vol').value = String(S.vol);
    document.getElementById('rg-sub').value = String(Math.round(S.sub * 100));
    document.getElementById('sel-liq').value = S.rhoL.toFixed(2);
    var modeBtn = document.getElementById('btn-mode');
    modeBtn.textContent = S.mode === 'hang' ? '模式：测力计悬挂' : '模式：自由浮沉';
    modeBtn.classList.toggle('is-on', S.mode === 'hang');
    document.getElementById('btn-soak').disabled = S.mode !== 'hang';
    document.getElementById('btn-step').textContent = S.mode === 'hang' ? '单步浸入' : '单步推进';
    var sub = document.getElementById('rg-sub');
    sub.disabled = S.mode !== 'hang';
  }

  document.getElementById('rg-rho').addEventListener('input', function () {
    S.rho = +this.value;
    if (S.mode === 'free') { S.yb = 0; S.v = 0; }
    render();
  });
  document.getElementById('rg-vol').addEventListener('input', function () {
    S.vol = +this.value;
    if (S.mode === 'free') { S.yb = 0; S.v = 0; }
    render();
  });
  document.getElementById('rg-sub').addEventListener('input', function () {
    S.sub = (+this.value) / 100;
    S.soak = false;
    render();
  });
  document.getElementById('sel-liq').addEventListener('change', function () {
    S.rhoL = +this.value;
    if (S.mode === 'free') { S.yb = 0; S.v = 0; }
    render();
  });

  document.getElementById('btn-mode').addEventListener('click', function () {
    S.mode = S.mode === 'hang' ? 'free' : 'hang';
    S.yb = 0;
    S.v = 0;
    S.soak = false;
    S.playing = S.mode === 'free';
    syncControls();
    render();
  });

  document.getElementById('btn-soak').addEventListener('click', function () {
    if (S.mode !== 'hang') return;
    S.sub = 0;
    S.soak = true;
    document.getElementById('rg-sub').value = '0';
    render();
  });

  document.getElementById('btn-step').addEventListener('click', function () {
    if (S.mode === 'hang') {
      // 往返步进：0→100% 逐步浸入，到顶后逐步提起，便于反复演示
      S.soak = false;
      S.sub += 0.1 * S.stepDir;
      if (S.sub >= 1 - 1e-9) { S.sub = 1; S.stepDir = -1; }
      if (S.sub <= 1e-9) { S.sub = 0; S.stepDir = 1; }
      document.getElementById('rg-sub').value = String(Math.round(S.sub * 100));
    } else {
      stepPhysics(0.4);
    }
    render();
  });

  document.getElementById('btn-record').addEventListener('click', record);
  document.getElementById('btn-csv').addEventListener('click', csv);

  document.getElementById('btn-reset').addEventListener('click', function () {
    S.rhoL = DEF.rhoL;
    S.rho = DEF.rho;
    S.vol = DEF.vol;
    S.sub = DEF.sub;
    S.mode = 'hang';
    S.yb = 0;
    S.v = 0;
    S.soak = false;
    S.stepDir = 1;
    S.playing = true;
    document.getElementById('rg-sub').value = String(Math.round(DEF.sub * 100));
    rowsEl.innerHTML = '';
    syncControls();
    render();
  });

  /* ---------------- 拖动旋转（伪 3D 视角） ---------------- */
  var drag = null;
  cv.addEventListener('pointerdown', function (e) {
    drag = { x: e.clientX, yaw: view.yaw };
    cv.setPointerCapture(e.pointerId);
    cv.style.cursor = 'grabbing';
  });
  cv.addEventListener('pointermove', function (e) {
    if (!drag) return;
    view.yaw = Math.max(-0.9, Math.min(0.9, drag.yaw + (e.clientX - drag.x) * 0.006));
    view.k = 0.26 - 0.055 * Math.abs(Math.sin(view.yaw * 0.9));
    render();
  });
  cv.addEventListener('pointerup', function () { drag = null; cv.style.cursor = 'grab'; });
  cv.addEventListener('pointercancel', function () { drag = null; cv.style.cursor = 'grab'; });
  cv.style.cursor = 'grab';

  /* ---------------- 启动 ---------------- */
  window.addEventListener('resize', function () { fit(); render(); });

  var last = performance.now();
  function loop(now) {
    var dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    S.t += dt;
    if (S.mode === 'free' && S.playing) stepPhysics(dt);
    if (S.soak) {
      S.sub += dt / 5;
      if (S.sub >= 1) { S.sub = 1; S.soak = false; }
      document.getElementById('rg-sub').value = String(Math.round(S.sub * 100));
    }
    render();
    requestAnimationFrame(loop);
  }

  fit();
  syncControls();
  render();
  requestAnimationFrame(loop);
})();
